```python
import statistics
import time

def benchmark(fn, inputs, repeats=5):
    """Return the median latency in milliseconds for each input."""
    results = {}
    for name, x in inputs.items():
        timings = []
        for _ in range(repeats):
            start = time.perf_counter()
            fn(x)
            timings.append((time.perf_counter() - start) * 1000)
        results[name] = statistics.median(timings)
    return results

if __name__ == "__main__":
    sizes = {f"n={n}": list(range(n, 0, -1)) for n in (1000, 10000, 100000)}
    for name, ms in benchmark(sorted, sizes).items():
        print(f"{name:<10} {ms:8.2f} ms")
```
